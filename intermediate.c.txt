#include <stdio.h>
#include <string.h>
#include <stdlib.h>

char op[5], arg1[10], arg2[10], result[10];

int main() {
    FILE *fp1 = fopen("input.txt", "r");
    FILE *fp2 = fopen("output.txt", "w");
    
    if (fp1 == NULL) {
        printf("Error: input.txt could not be opened.\n");
        return 1;
    }
    if (fp2 == NULL) {
        printf("Error: output.txt could not be created.\n");
        fclose(fp1);
        return 1;
    }
    
    while (fscanf(fp1, "%s%s%s%s", op, arg1, arg2, result) == 4) {
        if (strcmp(op, "+") == 0) {
            fprintf(fp2, "MOV R0, %s\n", arg1);
            fprintf(fp2, "ADD R0, %s\n", arg2);
            fprintf(fp2, "MOV %s, R0\n", result);
        } else if (strcmp(op, "*") == 0) {
            fprintf(fp2, "MOV R0, %s\n", arg1);
            fprintf(fp2, "MUL R0, %s\n", arg2);
            fprintf(fp2, "MOV %s, R0\n", result);
        } else if (strcmp(op, "-") == 0) {
            fprintf(fp2, "MOV R0, %s\n", arg1);
            fprintf(fp2, "SUB R0, %s\n", arg2);
            fprintf(fp2, "MOV %s, R0\n", result);
        } else if (strcmp(op, "/") == 0) {
            fprintf(fp2, "MOV R0, %s\n", arg1);
            fprintf(fp2, "DIV R0, %s\n", arg2);
            fprintf(fp2, "MOV %s, R0\n", result);
        } else if (strcmp(op, "=") == 0) {
            fprintf(fp2, "MOV R0, %s\n", arg1);
            fprintf(fp2, "MOV %s, R0\n", result);
        }
    }
    
    printf("Code generation complete. Output written to output.txt\n");
    fclose(fp1);
    fclose(fp2);
    return 0;
}
