#include <stdio.h>
#include <string.h>
#include <ctype.h>

int n, m = 0;
char a[20][20];
char f[50];

void addToResultSet(char val) {
    for (int k = 0; k < m; k++) {
        if (f[k] == val) return;
    }
    f[m++] = val;
}

void first(char c) {
    if (!isupper(c)) {
        addToResultSet(c);
        return;
    }
    for (int k = 0; k < n; k++) {
        if (a[k][0] == c) {
            if (a[k][2] == '$') {
                addToResultSet('$');
            } else if (!isupper(a[k][2])) {
                addToResultSet(a[k][2]);
            } else {
                first(a[k][2]);
            }
        }
    }
}

void follow(char c) {
    if (a[0][0] == c) {
        addToResultSet('$');
    }
    for (int i = 0; i < n; i++) {
        int len = strlen(a[i]);
        for (int j = 2; j < len; j++) {
            if (a[i][j] == c) {
                if (a[i][j + 1] != '\0') {
                    char next = a[i][j + 1];
                    if (!isupper(next)) {
                        addToResultSet(next);
                    } else {
                        int current_m = m;
                        first(next);
                        int has_epsilon = 0;
                        for (int k = current_m; k < m; k++) {
                            if (f[k] == '$') {
                                has_epsilon = 1;
                                f[k] = f[--m];
                                k--;
                            }
                        }
                        if (has_epsilon && c != a[i][0]) {
                            follow(a[i][0]);
                        }
                    }
                } else if (c != a[i][0]) {
                    follow(a[i][0]);
                }
            }
        }
    }
}

int main() {
    int z;
    char c;
    printf("Enter number of productions: ");
    if (scanf("%d", &n) != 1) return 0;
    
    printf("Enter productions (e.g., S=AB or S=a):\n");
    for (int i = 0; i < n; i++) {
        scanf("%s", a[i]);
    }
    
    do {
        m = 0;
        printf("\nEnter non-terminal to find First and Follow: ");
        scanf(" %c", &c);
        
        first(c);
        printf("First(%c) = { ", c);
        for (int i = 0; i < m; i++) printf("%c ", f[i]);
        printf("}\n");
        
        m = 0;
        follow(c);
        printf("Follow(%c) = { ", c);
        for (int i = 0; i < m; i++) printf("%c ", f[i]);
        printf("}\n");
        
        printf("Continue (1 to continue, 0 to exit)? ");
        scanf("%d", &z);
    } while (z == 1);
    
    return 0;
}
